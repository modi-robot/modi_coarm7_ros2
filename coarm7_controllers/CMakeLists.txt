cmake_minimum_required(VERSION 3.16)
project(coarm7_controllers LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(controller_interface REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(control_msgs REQUIRED)
find_package(realtime_tools REQUIRED)
find_package(pluginlib REQUIRED)
add_library(coarm7_endpoint_gripper_controller SHARED src/endpoint_gripper_controller.cpp)
target_compile_features(coarm7_endpoint_gripper_controller PUBLIC cxx_std_17)
target_include_directories(coarm7_endpoint_gripper_controller PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(coarm7_endpoint_gripper_controller controller_interface rclcpp rclcpp_action control_msgs realtime_tools pluginlib)
target_compile_options(coarm7_endpoint_gripper_controller PRIVATE -Wall -Wextra -Wpedantic)
pluginlib_export_plugin_description_file(controller_interface coarm7_controllers.xml)
install(TARGETS coarm7_endpoint_gripper_controller EXPORT export_coarm7_controllers LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
 find_package(ament_cmake_gtest REQUIRED)
 ament_add_gtest(test_endpoint_controller test/test_endpoint_controller.cpp)
 target_link_libraries(test_endpoint_controller coarm7_endpoint_gripper_controller)
 ament_target_dependencies(test_endpoint_controller controller_interface hardware_interface rclcpp rclcpp_action control_msgs realtime_tools)
endif()
ament_export_targets(export_coarm7_controllers HAS_LIBRARY_TARGET)
ament_export_dependencies(controller_interface rclcpp rclcpp_action control_msgs realtime_tools pluginlib)
ament_package()
