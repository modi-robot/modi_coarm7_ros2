cmake_minimum_required(VERSION 3.16)
project(coarm7_planning)
find_package(ament_cmake REQUIRED)
find_package(moveit_core REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(pilz_industrial_motion_planner REQUIRED)
add_library(coarm7_group_planner SHARED src/group_planner.cpp src/group_timing.cpp)
target_compile_features(coarm7_group_planner PUBLIC cxx_std_17)
target_include_directories(coarm7_group_planner PRIVATE include)
ament_target_dependencies(coarm7_group_planner moveit_core moveit_msgs pluginlib rclcpp)
pluginlib_export_plugin_description_file(moveit_core plugins.xml)
install(TARGETS coarm7_group_planner LIBRARY DESTINATION lib)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_routing test/test_routing.cpp)
  target_include_directories(test_routing PRIVATE include)
  ament_target_dependencies(test_routing moveit_msgs moveit_core)
endif()
ament_package()
