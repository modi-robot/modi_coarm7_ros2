cmake_minimum_required(VERSION 3.19)
project(coarm7_hardware LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(urdf REQUIRED)
find_package(spdlog REQUIRED)
find_package(Threads REQUIRED)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

option(COARM7_ENABLE_SDK_EXTENSIONS
  "Use a separately installed internal binary SDK extension package" OFF)
set(COARM7_INSTALL_TARGETS coarm7_hardware_core coarm7_hardware)
if(COARM7_ENABLE_SDK_EXTENSIONS)
  # Internal diagnostics may use a separately delivered binary provider.
  # No private SDK source/header snapshot is part of this repository.
  find_package(canevo_sdk_extensions CONFIG REQUIRED)
  set(COARM7_SDK_TARGET canevo_sdk_extensions::canevo_sdk_extensions)
  set(COARM7_SDK_MODE development_extensions)
  set(COARM7_SDK_EXTENSIONS_YAML true)
else()
  # Both architectures use this SDK package/target name. No SDK source or
  # internal header is compiled into the default driver.
  include(cmake/CoArm7SDK.cmake)
  set(COARM7_SDK_TARGET canevo_sdk_x86::canevo_sdk_x86)
  set(COARM7_SDK_MODE public)
  set(COARM7_SDK_EXTENSIONS_YAML false)
endif()
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/sdk_capabilities.yaml
  "api: ${COARM7_SDK_MODE}\nextensions: ${COARM7_SDK_EXTENSIONS_YAML}\nhost_trace_default: true\nusb_capture: true\n")

add_library(coarm7_hardware_core SHARED src/config.cpp src/engine.cpp src/gripper.cpp src/transports.cpp src/device.cpp)
target_include_directories(coarm7_hardware_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>
  )
target_link_libraries(coarm7_hardware_core PUBLIC yaml-cpp PRIVATE ${COARM7_SDK_TARGET} Threads::Threads)
target_compile_definitions(coarm7_hardware_core PUBLIC
  COARM7_SDK_EXTENSIONS=$<BOOL:${COARM7_ENABLE_SDK_EXTENSIONS}>)
target_compile_options(coarm7_hardware_core PRIVATE -Wall -Wextra -Wpedantic)
add_library(coarm7_hardware SHARED src/system.cpp)
target_include_directories(coarm7_hardware PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(coarm7_hardware hardware_interface pluginlib rclcpp rclcpp_lifecycle urdf)
target_link_libraries(coarm7_hardware coarm7_hardware_core)
target_compile_options(coarm7_hardware PRIVATE -Wall -Wextra -Wpedantic)
foreach(t ${COARM7_INSTALL_TARGETS})
  set_target_properties(${t} PROPERTIES INSTALL_RPATH "$ORIGIN" INSTALL_RPATH_USE_LINK_PATH TRUE)
endforeach()
add_executable(clear_drive_faults src/clear_drive_faults.cpp)
target_link_libraries(clear_drive_faults PRIVATE coarm7_hardware_core)
target_compile_options(clear_drive_faults PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(clear_drive_faults PROPERTIES INSTALL_RPATH "$ORIGIN/..")
install(TARGETS clear_drive_faults RUNTIME DESTINATION lib/${PROJECT_NAME})
pluginlib_export_plugin_description_file(hardware_interface coarm7_hardware.xml)
install(TARGETS ${COARM7_INSTALL_TARGETS}
  EXPORT export_coarm7_hardware LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/sdk_capabilities.yaml DESTINATION share/${PROJECT_NAME}/config)
install(FILES SDK_PROVENANCE.json DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_engine test/test_engine.cpp)
  target_link_libraries(test_engine coarm7_hardware_core)
  target_compile_definitions(test_engine PRIVATE CONFIG_PATH="${CMAKE_CURRENT_SOURCE_DIR}/config/offline.yaml")
  ament_add_gtest(test_system test/test_system.cpp)
  target_link_libraries(test_system coarm7_hardware)
  ament_target_dependencies(test_system hardware_interface rclcpp rclcpp_lifecycle)
  target_compile_definitions(test_system PRIVATE CONFIG_PATH="${CMAKE_CURRENT_SOURCE_DIR}/config/offline.yaml")
  ament_add_gtest(test_device test/test_device.cpp)
  target_link_libraries(test_device coarm7_hardware_core ${COARM7_SDK_TARGET})
  target_compile_definitions(test_device PRIVATE CONFIG_PATH="${CMAKE_CURRENT_SOURCE_DIR}/config/offline.yaml")
  ament_add_gtest(test_public_sdk test/test_public_sdk.cpp)
  target_link_libraries(test_public_sdk coarm7_hardware_core ${COARM7_SDK_TARGET})
  target_compile_definitions(test_public_sdk PRIVATE CONFIG_PATH="${CMAKE_CURRENT_SOURCE_DIR}/config/offline.yaml")
  ament_add_gtest(test_gripper_speed test/test_gripper_speed.cpp)
  target_link_libraries(test_gripper_speed coarm7_hardware_core)
  target_compile_definitions(test_gripper_speed PRIVATE CONFIG_PATH="${CMAKE_CURRENT_SOURCE_DIR}/config/offline.yaml")
  find_package(ament_cmake_pytest REQUIRED)
  get_filename_component(COARM7_HARDWARE_REAL_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}" REALPATH)
  ament_add_pytest_test(test_sdk_boundary "${COARM7_HARDWARE_REAL_SOURCE}/../tools/test_sdk_tools.py" TIMEOUT 45)
  ament_add_pytest_test(test_gripper_following test/test_gripper_following.py TIMEOUT 45)
endif()
ament_export_targets(export_coarm7_hardware HAS_LIBRARY_TARGET)
ament_export_dependencies(hardware_interface pluginlib rclcpp rclcpp_lifecycle yaml-cpp urdf)
if(COARM7_ENABLE_SDK_EXTENSIONS)
  ament_export_dependencies(canevo_sdk_extensions)
else()
  ament_export_dependencies(canevo_sdk_x86)
endif()
ament_package()
