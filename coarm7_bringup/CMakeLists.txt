cmake_minimum_required(VERSION 3.16)
project(coarm7_bringup)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(moveit_ros_planning REQUIRED)
find_package(random_numbers REQUIRED)
add_executable(coarm7_model_check src/model_check.cpp)
target_compile_features(coarm7_model_check PUBLIC cxx_std_17)
ament_target_dependencies(coarm7_model_check rclcpp moveit_ros_planning random_numbers)
install(TARGETS coarm7_model_check DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/validate_mock.py scripts/start_coarm7.py scripts/fetch_robot_model.py scripts/affinity_guard.py scripts/validate_offline.py scripts/validate_ptp.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch rviz DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
 find_package(ament_cmake_pytest REQUIRED)
 ament_add_pytest_test(test_startup test/test_startup.py TIMEOUT 45)
 ament_add_pytest_test(test_effective_limits test/test_effective_limits.py TIMEOUT 60)
 ament_add_pytest_test(test_affinity test/test_affinity.py TIMEOUT 45)
 ament_add_pytest_test(test_architecture test/test_architecture.py TIMEOUT 60)
endif()
ament_package()
